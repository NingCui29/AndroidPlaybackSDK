cmake_minimum_required(VERSION 3.22)
project(playback LANGUAGES C)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_library(playback SHARED playback_input.c playback_jni.c software_decoder.c audio_output.c video_output.c packet_queue.c decode_tracks.c dual_output.c session_output.c media_info.c recovery_gop.c)

target_compile_options(playback PRIVATE -Wall -Werror)

target_link_directories(playback PRIVATE "${ROOT}/artifacts/n01/stripped/${ANDROID_ABI}")
target_link_libraries(playback PRIVATE avformat avcodec avutil swresample swscale android m)
